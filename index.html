# الصورة البرج الدور رقم اللوحة المدينة التاريخ الوقت ملاحظات رجل الأمن

⚙️ كود Google Apps Script المطلوب

انسخ هذا الكود بالكامل والصقه في Google Apps Script بدلاً من الكود القديم، ثم انشر:

1 افتح Apps Script وامسح الكود القديم وضع هذا بدله:

// ===== كود Google Apps Script الكامل =====

// الصق هذا في Apps Script بدلاً من الكود القديم

var SHEET_NAME = ""; // اتركه فارغاً لاستخدام الشيت الأول

function doGet(e) {
  var params = e.parameter;
  var callback = params.callback;
  var action = params.action || "getAll";

  if (action === "add") {
    return handleAdd(params, callback);
  }

  if (action === "getAll") {
    return handleGetAll(callback);
  }

  return respond({error: "unknown action"}, callback);
}

function handleAdd(params, callback) {
  var sheet = getSheet();

  // إنشاء رؤوس الأعمدة تلقائياً إن لم تكن موجودة
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      "البرج", "الدور", "رقم اللوحة", "المدينة",
      "التاريخ", "الوقت", "ملاحظات",
      "رجل الأمن", "الرقم الوظيفي", "صورة", "وقت الإضافة"
    ]);
  }

  sheet.appendRow([
    params.tower    || "",
    params.floor    || "",
    params.plate    || "",
    params.city     || "",
    params.date     || "",
    params.time     || "",
    params.notes    || "",
    params.empName  || "",
    params.empId    || "",
    params.hasImage || "لا",
    new Date().toISOString()
  ]);

  return respond({success: true, message: "تم الحفظ"}, callback);
}

function handleGetAll(callback) {
  var sheet = getSheet();
  var data = sheet.getDataRange().getValues();

  if (data.length <= 1) {
    return respond([], callback);
  }

  var headers = data[0];
  var rows = data.slice(1).map(function(row) {
    var obj = {};
    headers.forEach(function(h, i) {
      obj[String(h)] = row[i];
    });
    return obj;
  });

  return respond(rows, callback);
}

function getSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  return SHEET_NAME ? ss.getSheetByName(SHEET_NAME) : ss.getActiveSheet();
}

function respond(data, callback) {
  var json = JSON.stringify(data);
  var output = callback ? callback + "(" + json + ");" : json;
  return ContentService
    .createTextOutput(output)
    .setMimeType(
      callback
        ? ContentService.MimeType.JAVASCRIPT
        : ContentService.MimeType.JSON
    );
}

// دالة doPost للتوافق (اختياري)
function doPost(e) {
  var params = JSON.parse(e.postData.contents || "{}");
  return handleAdd(params, null);
}
