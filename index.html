انسخ وضعه في Apps Script بدلاً من الكود القديم ثم انشر:

1 الصق الكود:

var SHEET_NAME = "رصد السيارات";

function doGet(e) {
var params = e.parameter;
var cb = params.callback;
var action = params.action || “getAll”;
if (action === “add”)    return handleAdd(params, cb);
if (action === “getAll”) return handleGetAll(cb);
return respond({error:“unknown”}, cb);
}

function handleAdd(p, cb) {
var ss = SpreadsheetApp.getActiveSpreadsheet();
var sheet = getOrCreate(ss);
if (sheet.getLastRow() === 0) setupHeaders(sheet);
var nextRow = sheet.getLastRow() + 1;
sheet.getRange(nextRow,1,1,10).setValues([[
p.tower    || “”,
p.floor    || “”,
p.plate    || “”,
p.obsDate  || p.date || “”,
p.time     || “”,
p.notes    || “”,
p.empName  || “”,
p.empId    || “”,
p.hasImage || “لا”,
new Date().toLocaleString(“ar-SA”)
]]);
formatRow(sheet, nextRow);
return respond({success:true}, cb);
}

function handleGetAll(cb) {
var sheet = getSheet();
if (!sheet) return respond([], cb);
var data = sheet.getDataRange().getValues();
if (data.length <= 1) return respond([], cb);
var headers = data[0];
var rows = data.slice(1).map(function(row) {
var obj = {};
headers.forEach(function(h,i){ obj[String(h)] = row[i]; });
return obj;
});
return respond(rows, cb);
}

function setupHeaders(sheet) {
var h = [“🏢 البرج”,“🔢 الدور”,“🚗 رقم اللوحة”,
“📅 تاريخ الرصد”,“⏰ الوقت”,“📝 ملاحظات”,
“👤 رجل الأمن”,“🪪 الرقم الوظيفي”,“📷 صورة”,“🕐 وقت الإضافة”];
var r = sheet.getRange(1,1,1,h.length);
r.setValues([h])
.setBackground(”#1e3a8a”).setFontColor(”#ffffff”)
.setFontWeight(“bold”).setFontSize(11)
.setHorizontalAlignment(“center”);
sheet.setRowHeight(1, 36);
[70,70,130,110,75,160,140,100,70,155].forEach(function(w,i){
sheet.setColumnWidth(i+1, w);
});
sheet.setFrozenRows(1);
sheet.setRightToLeft(true);
}

function formatRow(sheet, row) {
var r = sheet.getRange(row,1,1,10);
r.setBackground(row%2===0?”#f0f4ff”:”#ffffff”)
.setFontSize(10).setVerticalAlignment(“middle”);
sheet.setRowHeight(row, 30);
sheet.getRange(row,3)
.setBackground(”#1e3a8a”).setFontColor(”#ffffff”)
.setFontWeight(“bold”).setHorizontalAlignment(“center”);
sheet.getRange(row,1)
.setBackground(”#dbeafe”).setFontColor(”#1d4ed8”)
.setFontWeight(“bold”).setHorizontalAlignment(“center”);
sheet.getRange(row,2)
.setBackground(”#fef9c3”).setFontColor(”#854d0e”)
.setFontWeight(“bold”).setHorizontalAlignment(“center”);
sheet.getRange(row,4,1,2).setHorizontalAlignment(“center”);
r.setBorder(true,true,true,true,true,true,”#e2e8f0”,
SpreadsheetApp.BorderStyle.SOLID);
}

function getOrCreate(ss) {
return ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);
}
function getSheet() {
var ss = SpreadsheetApp.getActiveSpreadsheet();
return SHEET_NAME ? ss.getSheetByName(SHEET_NAME) : ss.getActiveSheet();
}
function respond(data, cb) {
var json = JSON.stringify(data);
var out = cb ? cb+”(”+json+”);” : json;
return ContentService.createTextOutput(out).setMimeType(
cb ? ContentService.MimeType.JAVASCRIPT : ContentService.MimeType.JSON);
}
function doPost(e) {
return handleAdd(JSON.parse(e.postData.contents||”{}”), null);
}
function formatAllExistingRows() {
var sheet = getSheet(); if(!sheet) return;
for(var r=2;r<=sheet.getLastRow();r++) formatRow(sheet,r);
SpreadsheetApp.flush();
}
