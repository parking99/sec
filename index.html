⚙️ إعداد Google Apps Script

انسخ الكود والصقه في Apps Script ثم انشر من جديد. تأكد من إعداد ورقة الموظفين بشكل صحيح.

📋 هيكل ورقة الموظفين

أنشئ ورقة باسم الموظفون بهذه الأعمدة:
A: الرقم الوظيفي B: الاسم C: القسم D: كلمة السر E: الدور (اختياري)

1 انسخ الكود الكامل: 
const SHEET_NAME   = "رصد السيارات";
const GUARDS_SHEET = "الموظفون";
const DRIVE_FOLDER = "صور مركبات الرصد";

function doGet(e) {
  const p      = (e && e.parameter) ? e.parameter : {};
  const cb     = p.callback || "";
  const action = p.action   || "getAll";
  if (action === "add")         return handleAdd(p, cb);
  if (action === "getAll")      return handleGetAll(cb);
  if (action === "getGuard")    return handleGetGuard(p, cb);
  if (action === "login")       return handleLogin(p, cb);
  if (action === "uploadImage") return handleImageUpload(p, cb);
  return makeResponse({ error: "unknown_action" }, cb);
}

function doPost(e) {
  try {
    const body = JSON.parse((e && e.postData && e.postData.contents) ? e.postData.contents : "{}");
    if (body.action === "uploadImage") return handleImageUpload(body);
    return handleAdd(body, "");
  } catch (err) {
    return makeResponse({ error: String(err && err.message ? err.message : err) }, "");
  }
}

function getSheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  return ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);
}

function setupHeaders_(sheet) {
  const h = ["البرج","الدور","رقم اللوحة","تاريخ الرصد","الوقت","حالة المركبة","ملاحظات","رجل الأمن","الرقم الوظيفي","صورة","وقت الإضافة","رابط الصورة","رقم الموقف","رقم العامود"];
  sheet.getRange(1,1,1,h.length).setValues([h]).setBackground("#0f172a").setFontColor("#ffffff").setFontWeight("bold").setFontSize(11).setHorizontalAlignment("center");
  sheet.setRowHeight(1,40); sheet.setFrozenRows(1); sheet.setRightToLeft(true);
}

function formatRow_(sheet, row) {
  sheet.getRange(row,1,1,12).setBackground(row%2===0?"#f8fafc":"#ffffff").setFontSize(10).setVerticalAlignment("middle");
  sheet.setRowHeight(row,32);
  const sv = String(sheet.getRange(row,6).getValue()||"").trim();
  if(sv==="مغبرة") sheet.getRange(row,6).setBackground("#fee2e2").setFontColor("#991b1b").setFontWeight("bold");
  else if(sv==="متوسطة الغبار") sheet.getRange(row,6).setBackground("#fef9c3").setFontColor("#854d0e").setFontWeight("bold");
  else if(sv==="حديثة") sheet.getRange(row,6).setBackground("#dcfce7").setFontColor("#166534").setFontWeight("bold");
}

function makeResponse(data, cb) {
  const json = JSON.stringify(data);
  const out  = (cb&&cb!=="")?(cb+"("+json+");"):json;
  const mime = (cb&&cb!=="") ? ContentService.MimeType.JAVASCRIPT : ContentService.MimeType.JSON;
  return ContentService.createTextOutput(out).setMimeType(mime);
}

function handleLogin(p, cb) {
  const id  = String(p.empId||"").trim();
  const pwd = String(p.password||"").trim();
  if(!id || !pwd) return makeResponse({error:"missing_fields"}, cb);
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(GUARDS_SHEET);
  if(!sheet) return makeResponse({error:"no_sheet", msg:"أنشئ ورقة: "+GUARDS_SHEET}, cb);
  const rows = sheet.getDataRange().getValues();
  for(let i=1; i<rows.length; i++) {
    if(String(rows[i][0]).trim() === id) {
      const storedPwd = String(rows[i][3]||"").trim();
      if(storedPwd === pwd) {
        return makeResponse({
          success: true,
          empId: id,
          name:  String(rows[i][1]||""),
          dept:  String(rows[i][2]||""),
          role:  String(rows[i][4]||"guard")
        }, cb);
      } else {
        return makeResponse({error:"wrong_password"}, cb);
      }
    }
  }
  return makeResponse({error:"not_found"}, cb);
}

function handleAdd(p, cb) {
  const sheet = getSheet_();
  if(sheet.getLastRow()===0) setupHeaders_(sheet);
  const nextRow = sheet.getLastRow()+1;
  sheet.getRange(nextRow,1,1,14).setValues([[p.tower||"",p.floor||"",p.plate||"",p.obsDate||p.date||"",p.time||"",p.carStatus||"",p.notes||"",p.empName||"",p.empId||"",p.hasImage||"لا",new Date().toLocaleString("ar-SA"),"",p.parkingNum||"",p.colNum||""]]);
  formatRow_(sheet, nextRow);
  SpreadsheetApp.flush();
  return makeResponse({success:true, row:nextRow}, cb);
}

function handleGetAll(cb) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  if(!sheet) return makeResponse([],cb);
  const data = sheet.getDataRange().getValues();
  if(data.length<=1) return makeResponse([],cb);
  const headers = data[0], tz = Session.getScriptTimeZone();
  const rows = data.slice(1).map(r => {
    const obj = {};
    headers.forEach((h,i) => {
      let val = r[i];
      if(val instanceof Date) {
        if(val.getFullYear()<=1900){const hh=val.getHours(),mm=val.getMinutes();val=(hh<10?"0":"")+hh+":"+(mm<10?"0":"")+mm;}
        else val = Utilities.formatDate(val,tz,"yyyy-MM-dd");
      }
      obj[String(h)] = val;
    });
    return obj;
  });
  return makeResponse(rows,cb);
}

function handleGetGuard(p, cb) {
  const id = String(p.empId||"").trim();
  if(!id) return makeResponse({error:"no_id"},cb);
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(GUARDS_SHEET);
  if(!sheet) return makeResponse({error:"no_sheet",msg:"أنشئ ورقة: "+GUARDS_SHEET},cb);
  const rows = sheet.getDataRange().getValues();
  for(let i=1;i<rows.length;i++) {
    if(String(rows[i][0]).trim()===id) return makeResponse({success:true,empId:id,name:String(rows[i][1]||""),dept:String(rows[i][2]||"")},cb);
  }
  return makeResponse({error:"not_found"},cb);
}

function handleImageUpload(p, cb) {
  cb = cb||"";
  try {
    const plate=String(p.plate||"unknown").replace(/[^A-Za-z0-9]/g,"").toUpperCase();
    const date=String(p.date||"").trim(), time=String(p.time||"").trim();
    const imgData=String(p.imgData||""), rowHint=Number(p.row||0)||0;
    if(!imgData) return makeResponse({error:"no_image"},"");
    const base64 = imgData.indexOf(",")>-1 ? imgData.split(",")[1] : imgData;
    const filename=[plate,date,time?time.replace(":","-"):""].filter(Boolean).join("_")+".jpg";
    const blob = Utilities.newBlob(Utilities.base64Decode(base64),"image/jpeg",filename);
    const folders=DriveApp.getFoldersByName(DRIVE_FOLDER);
    const folder=folders.hasNext()?folders.next():DriveApp.createFolder(DRIVE_FOLDER);
    const file=folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK,DriveApp.Permission.VIEW);
    const fileId=file.getId();
    const viewUrl="https://drive.google.com/uc?export=view&id="+fileId;
    const thumbUrl="https://drive.google.com/thumbnail?id="+fileId+"&sz=w800";
    updateImageUrl_(plate,date,time,thumbUrl,rowHint);
    return makeResponse({success:true,viewUrl,thumbUrl,fileId},cb);
  } catch(err) {
    return makeResponse({error:String(err&&err.message?err.message:err)},cb);
  }
}

function updateImageUrl_(plate,date,time,url,rowHint) {
  const sheet=SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  if(!sheet) return;
  const data=sheet.getDataRange().getValues();
  if(rowHint&&rowHint>=2&&rowHint<=data.length){sheet.getRange(rowHint,12).setValue(url);sheet.getRange(rowHint,10).setValue("نعم");return;}
  for(let r=data.length;r>=2;r--){
    const p=String(data[r-1][2]||"").trim().toUpperCase();
    const d=String(data[r-1][3]||"").trim();
    const t=String(data[r-1][4]||"").trim();
    if(p===plate&&(!date||d===date)&&(!time||t===time)){sheet.getRange(r,12).setValue(url);sheet.getRange(r,10).setValue("نعم");return;}
  }
}
