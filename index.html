⚙️ كود Google Apps Script المطلوب

انسخ هذا الكود بالكامل والصقه في Apps Script بدلاً من الكود القديم، ثم انشر:

1 امسح الكود القديم وضع هذا بدله:

var SHEET_NAME = "رصد السيارات"; // اسم الشيت

// ══════════════════════════════════════════
//   رصد السيارات — Google Apps Script
//   نسخة منظّمة مع تنسيق احترافي
// ══════════════════════════════════════════

function doGet(e) {
var params   = e.parameter;
var callback = params.callback;
var action   = params.action || "getAll";

if (action === "add")    return handleAdd(params, callback);
if (action === "getAll") return handleGetAll(callback);
return respond({ error: "unknown action" }, callback);
}

// ── إضافة سجل جديد ──
function handleAdd(params, callback) {
var ss    = SpreadsheetApp.getActiveSpreadsheet();
var sheet = getOrCreateSheet(ss);

// إنشاء الرؤوس إن لم تكن موجودة
if (sheet.getLastRow() === 0) {
setupHeaders(sheet);
}

var nextRow = sheet.getLastRow() + 1;

// كتابة البيانات
sheet.getRange(nextRow, 1, 1, 11).setValues([[
params.tower    || "",
params.floor    || "",
params.plate    || "",
params.city     || "",
params.date     || "",
params.time     || "",
params.notes    || "",
params.empName  || "",
params.empId    || "",
params.hasImage || "لا",
new Date().toLocaleString("ar-SA")
]]);

// تنسيق الصف الجديد
formatDataRow(sheet, nextRow);

return respond({ success: true }, callback);
}

// ── جلب جميع البيانات ──
function handleGetAll(callback) {
var sheet = getSheet();
if (!sheet) return respond([], callback);

var data = sheet.getDataRange().getValues();
if (data.length <= 1) return respond([], callback);

var headers = data[0];
var rows = data.slice(1).map(function(row) {
var obj = {};
headers.forEach(function(h, i) { obj[String(h)] = row[i]; });
return obj;
});

return respond(rows, callback);
}

// ══ إنشاء / تنسيق الرؤوس ══
function setupHeaders(sheet) {
var headers = [
"🏢 البرج", "🔢 الدور", "🚗 رقم اللوحة", "📍 المدينة",
"📅 التاريخ", "⏰ الوقت", "📝 ملاحظات",
"👤 رجل الأمن", "🪪 الرقم الوظيفي", "📷 صورة", "🕐 وقت الإضافة"
];

var headerRange = sheet.getRange(1, 1, 1, headers.length);
headerRange.setValues([headers]);

// تنسيق الرؤوس
headerRange
.setBackground("#1e3a8a")       // أزرق داكن
.setFontColor("#ffffff")        // نص أبيض
.setFontWeight("bold")
.setFontSize(11)
.setHorizontalAlignment("center")
.setVerticalAlignment("middle");

sheet.setRowHeight(1, 36);

// عرض الأعمدة
var widths = [70, 70, 130, 100, 100, 80, 160, 140, 110, 70, 160];
widths.forEach(function(w, i) {
sheet.setColumnWidth(i + 1, w);
});

// تجميد الصف الأول
sheet.setFrozenRows(1);

// اتجاه RTL
sheet.setRightToLeft(true);
}

// ══ تنسيق صف البيانات ══
function formatDataRow(sheet, rowIndex) {
var range = sheet.getRange(rowIndex, 1, 1, 11);

// تبادل الألوان: فردي / زوجي
var bgColor = (rowIndex % 2 === 0) ? "#f0f4ff" : "#ffffff";
range.setBackground(bgColor);
range.setFontSize(10);
range.setVerticalAlignment("middle");
sheet.setRowHeight(rowIndex, 30);

// رقم اللوحة — تمييز خاص
var plateCell = sheet.getRange(rowIndex, 3);
plateCell
.setBackground("#1e3a8a")
.setFontColor("#ffffff")
.setFontWeight("bold")
.setFontSize(11)
.setHorizontalAlignment("center");

// عمود البرج
sheet.getRange(rowIndex, 1)
.setBackground("#dbeafe")
.setFontColor("#1d4ed8")
.setFontWeight("bold")
.setHorizontalAlignment("center");

// عمود الدور
sheet.getRange(rowIndex, 2)
.setBackground("#fef9c3")
.setFontColor("#854d0e")
.setFontWeight("bold")
.setHorizontalAlignment("center");

// عمود المدينة
sheet.getRange(rowIndex, 4)
.setBackground("#dcfce7")
.setFontColor("#166534")
.setHorizontalAlignment("center");

// التاريخ والوقت
sheet.getRange(rowIndex, 5, 1, 2)
.setHorizontalAlignment("center")
.setFontColor("#475569");

// صورة
sheet.getRange(rowIndex, 10)
.setHorizontalAlignment("center");

// حدود خفيفة لكل الخلايا
range.setBorder(
true, true, true, true, true, true,
"#e2e8f0",
SpreadsheetApp.BorderStyle.SOLID
);
}

// ══ دوال مساعدة ══
function getOrCreateSheet(ss) {
var sheet = ss.getSheetByName(SHEET_NAME);
if (!sheet) {
sheet = ss.insertSheet(SHEET_NAME);
}
return sheet;
}

function getSheet() {
var ss = SpreadsheetApp.getActiveSpreadsheet();
return SHEET_NAME ? ss.getSheetByName(SHEET_NAME) : ss.getActiveSheet();
}

function respond(data, callback) {
var json = JSON.stringify(data);
var out  = callback ? callback + "(" + json + ");" : json;
return ContentService.createTextOutput(out)
.setMimeType(
callback
? ContentService.MimeType.JAVASCRIPT
: ContentService.MimeType.JSON
);
}

function doPost(e) {
var p = JSON.parse(e.postData.contents || "{}");
return handleAdd(p, null);
}

// ══ تنسيق جميع البيانات الموجودة (شغّلها مرة واحدة) ══
function formatAllExistingRows() {
var sheet = getSheet();
if (!sheet) return;
var lastRow = sheet.getLastRow();
if (lastRow <= 1) return;
for (var r = 2; r <= lastRow; r++) {
formatDataRow(sheet, r);
}
SpreadsheetApp.flush();
Logger.log("تم تنسيق " + (lastRow - 1) + " صف");
}
